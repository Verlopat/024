cmake_minimum_required(VERSION 3.18)
project(SAW_DSU LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(dsu_bench src/dsu.cpp)
target_compile_options(dsu_bench PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(dsu_bench PRIVATE Threads::Threads)
add_executable(dsu_harness verify/harness.cpp)
target_link_libraries(dsu_harness PRIVATE Threads::Threads)
enable_testing()
add_test(NAME dsu_harness COMMAND dsu_harness)
